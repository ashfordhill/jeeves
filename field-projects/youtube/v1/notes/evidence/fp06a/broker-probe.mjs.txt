const confirm=(queue,body,id)=>new Promise((resolve,reject)=>channel.sendToQueue(queue,Buffer.from(body),{messageId:id,persistent:true},e=>e?reject(e):resolve()));

 const fc=await consume(prefix+'-fatal',1,0,async()=>{throw new NonRetryable('INVALID_MEDIA');},{onTerminal:async()=>{throw new Error('simulated hook failure');}});await confirm(prefix+'-fatal','fatal',randomUUID());assert.deepEqual(await Promise.race([fc.closed,new Promise((_,reject)=>setTimeout(()=>reject(new Error('Fatal close deadline')),5000))]),{fatal:true});process.exitCode=0;const requeued=await channel.get(prefix+'-fatal',{noAck:true});assert.ok(requeued);assert.equal(requeued.fields.redelivered,true);

}finally{for(const q of queues)await channel.deleteQueue(q);await channel.close();await connection.close();}
