import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { setTimeout as delay } from "node:timers/promises";
import test from "node:test";
const env = process.env;
assert.equal(env.JEEVES_TEST_ISOLATED, "1");
const compose = (...args) => execFileSync("docker", ["compose", "--project-directory", env.JEEVES_TEST_PROJECT_DIR, "--env-file", "/dev/null", "-p", env.JEEVES_TEST_COMPOSE_PROJECT, "-f", env.JEEVES_TEST_COMPOSE_FILE, ...args], { cwd: env.JEEVES_TEST_PROJECT_DIR, encoding: "utf8", timeout: 120000 });
const base = () => {
  const address = compose("port", "orders", env.JEEVES_TEST_SERVICE_PORT).trim();
  assert.match(address, /^127\.0\.0\.1:\d+$/);
  return `http://${address}`;
};
test("ORDER-01: generated transactional event executes once and survives service/database restart", async () => {
  let url = base();
  const produce = async (body) => {
    const response = await fetch(`${url}/proof/orders`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
    assert.equal(response.status, 202); return (await response.json()).id;
  };
  const command = { key: "order-123", orderId: "order-123", quantity: 3 };
  const id = await produce(command);
  assert.equal(await produce(command), id);
  let result;
  for (let i = 0; i < 80; i++) {
    result = await (await fetch(`${url}/proof/events/${id}`)).json();
    if (result.status.state === "succeeded") break;
    await delay(100);
  }
  assert.equal(result.status.state, "succeeded");
  assert.equal(result.status.effectCommitted, true);
  assert.equal(result.status.attempts, 1);
  assert.deepEqual(result.reservations, [{ tenant_id: "fixture-tenant", order_id: "order-123", quantity: 3 }]);
  assert.equal("payload" in result.status, false);
  assert.equal("token" in result.status, false);
  compose("stop", "orders");
  compose("restart", "orders-db");
  compose("up", "-d", "--wait", "--wait-timeout", "90");
  url = base();
  assert.equal(await produce(command), id);
  const restored = await (await fetch(`${url}/proof/events/${id}`)).json();
  assert.deepEqual(restored, result);
  const second = await produce({ key: "order-124", orderId: "order-124", quantity: 1 });
  for (let i = 0; i < 80; i++) {
    const next = await (await fetch(`${url}/proof/events/${second}`)).json();
    if (next.status.state === "succeeded") { assert.equal(next.reservations.length, 1); return; }
    await delay(100);
  }
  assert.fail("Worker did not resume after restart");
});
