// Disposable conformance consumer only. Never included in generated application defaults.
import { defineFeature } from "@jeeves/service-runtime";
let slowStarted = false;
let hangOnStop = false;
export default defineFeature({
  name: "execution-proof",
  register(app) {
    app.post("/proof/crash", async () => {
      setTimeout(() => process.exit(42), 100);
      return { scheduled: true };
    });
    app.get("/proof/slow", async () => {
      slowStarted = true;
      await new Promise(resolve => setTimeout(resolve, 2000));
      return { completed: true };
    });
    app.get("/proof/state", async () => ({ slowStarted }));
    app.post("/proof/hang-on-stop", async () => { hangOnStop = true; return { armed: true }; });
  },
  async stop() {
    if (hangOnStop) {
      // Model a stuck external handle; an unresolved Promise alone does not keep Node alive.
      setInterval(() => {}, 1000);
      await new Promise(() => {});
    }
  },
});
