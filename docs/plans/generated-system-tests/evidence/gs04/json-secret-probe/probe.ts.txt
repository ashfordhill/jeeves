import assert from 'node:assert/strict';
import {mkdtemp,mkdir,writeFile,readFile,access,rm,chmod} from 'node:fs/promises';
import {tmpdir} from 'node:os';
import {join} from 'node:path';
import {randomUUID} from 'node:crypto';
import {sanitizer} from '/home/ash/github/jeeves/jeeves-generator/integration/harness/evidence.ts';
import {cleanupRun} from '/home/ash/github/jeeves/jeeves-generator/integration/harness/cleanup.ts';
async function main(){
