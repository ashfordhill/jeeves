import { readFile, writeFile } from 'node:fs/promises';
import { execFileSync } from 'node:child_process';
import { studioStudySeedOperations } from '/home/ash/github/jeeves/jeeves/packages/spec/src/index.ts';
import { frontDoorDesign } from '/home/ash/github/jeeves/field-projects/generator-system/fixtures/wd-front-door/design.ts';
import { test, expect, tile, createStudy, ready, showView } from '/home/ash/github/jeeves/jeeves/apps/web/e2e/studio.ts';
test('WD14 TinyURL renders and mouse export links the exact deployment inventory',async({page,tinyurl})=>{
 await showView(page,'Simple');
 await expect(tile(page,'Redirect service')).toBeVisible();
 await expect(page.locator('.board .board-wire')).toHaveCount(0);
 await expect(page.locator('.board .react-flow__edge')).toHaveCount(12);
 await page.getByRole('button',{name:'Fit',exact:true}).click();
 await page.screenshot({path:'/tmp/jeeves-wd14/tinyurl-mouse.png'});
 const id=await createStudy(page,'E2E WD14 README export','blank');
 try{
  const source=frontDoorDesign('required');
  const ops=studioStudySeedOperations(source);
  ops.push({op:'setGeneration',generation:source.generation!});
  const seed=await page.request.post(`/api/projects/${id}/operations`,{data:{baseRevision:0,source:'api',ops}});
  expect(seed.ok(),await seed.text()).toBe(true);
  await page.reload();await ready(page);
  await tile(page,'Public API').locator('.lane-container-header').click();
  await expect(page.locator('.studio-inspector').getByLabel('Front door',{exact:true})).toHaveValue('required');
  await page.getByRole('button',{name:'Generate project',exact:true}).click();
  const preview=page.waitForResponse(r=>r.url().endsWith('/experimental-generation/preview'));
  await page.getByRole('button',{name:'Preview scaffold',exact:true}).click();
  expect((await(await preview).json()).report.findings).toEqual([]);
  const event=page.waitForEvent('download');
  await page.getByRole('button',{name:'Download project ZIP',exact:true}).click();
  const download=await event;
  const path='/tmp/jeeves-wd14/mouse-export.zip';await download.saveAs(path);
  const archive=(name:string)=>execFileSync('python3',['-c','import sys,zipfile;sys.stdout.buffer.write(zipfile.ZipFile(sys.argv[1]).read(sys.argv[2]))',path,name],{encoding:'utf8'});
  const root=archive('README.md');
  const guide=archive('deploy/README.md');
  const contract=JSON.parse(archive('deploy/contract.json'));
  expect(root).toContain('**Local configuration summary.**');
  expect(root).toContain('[deployment guide](deploy/README.md)');
  expect(root).toContain('[machine-readable contract](deploy/contract.json)');
  expect(root).toContain('- `app-front`: none');
  expect(guide).toContain('- Front door: require x-edge-stamp');
  expect(contract.processes[0].env).toEqual(['JEEVES_EDGE_STAMP_KEY','JEEVES_EDGE_STAMP_KEY_PREVIOUS']);
  await expect(page.getByText(/Project ZIP downloaded/)).toBeVisible();
  await page.screenshot({path:'/tmp/jeeves-wd14/mouse-export.png'});
  await writeFile('/tmp/jeeves-wd14/browser-receipt.json',JSON.stringify({tinyurl:{rendered:true,legacyEdges:12,wires:0,mouseFit:true},export:{mousePreview:true,mouseDownload:true,limitedSummary:true,deploymentLinks:true,exactStampInventory:contract.processes[0].env},projectCleanedInFinally:id},null,2)+'\n');
 }finally{await page.request.delete(`/api/projects/${id}`);}
});
